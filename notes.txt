# docker-compose.prod.yml - Production configuration

services:
  database:
    image: postgres:15-alpine
    container_name: insider-threat-db-prod
    environment:
      POSTGRES_DB: insider_threat
      POSTGRES_USER: ${DATABASE_USERNAME:-postgres}
      POSTGRES_PASSWORD: ${DATABASE_PASSWORD:-postgres}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    networks:
      - app-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${DATABASE_USERNAME:-postgres} -d insider_threat"]
      interval: 30s
      timeout: 10s
      retries: 5

  backend:
    build: 
      context: ./backend
      dockerfile: Dockerfile
      target: production
    container_name: insider-threat-backend-prod
    environment:
      # Database connection
      DATABASE_HOST: database
      DATABASE_PORT: 5432
      DATABASE_USERNAME: ${DATABASE_USERNAME:-postgres}
      DATABASE_PASSWORD: ${DATABASE_PASSWORD:-postgres}
      DATABASE_NAME: insider_threat
      
      # Node.js settings
      NODE_ENV: production
      PORT: 3000
      
      # JWT Configuration - MUST be set securely in production
      JWT_SECRET: ${JWT_SECRET}
      JWT_EXPIRES_IN: ${JWT_EXPIRES_IN:-24h}
      
      # CORS settings for production
      FRONTEND_URL: ${FRONTEND_URL:-https://yourdomain.com}
      
      # Storage paths
      CSV_DROP_PATH: /app/storage/csv/drop
      CSV_PROCESSED_PATH: /app/storage/csv/processed
      CSV_ERROR_PATH: /app/storage/csv/error
      CSV_MONITOR_INTERVAL: 10000
    ports:
      - "3000:3000"
    depends_on:
      database:
        condition: service_healthy
    restart: unless-stopped
    volumes:
      - ./backend/storage:/app/storage
    networks:
      - app-network
    healthcheck:
      test: ["CMD", "wget", "--no-verbose", "--tries=1", "--spider", "http://localhost:3000/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 30s  # Give the application time to start up

  frontend:
    build: 
      context: ./frontend
      dockerfile: Dockerfile
    container_name: insider-threat-frontend-prod
    environment:
      NEXT_PUBLIC_API_URL: ${NEXT_PUBLIC_API_URL:-http://localhost:3000}
      NODE_ENV: production
      PORT: 3000
    ports:
      - "3001:3000"
    depends_on:
      backend:
        condition: service_healthy
    restart: unless-stopped
    networks:
      - app-network

volumes:
  postgres_data:

networks:
  app-network:
    driver: bridge
